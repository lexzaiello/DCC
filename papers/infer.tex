    { }
    {\ast : \top} \\\\
    \multicolumn{1}{l}{
    \inferrule[refl]
    {B : A \arr \sort{u}}
    {\text{refl}(B) : (x : !(A)) \arr (y : (!\la B \ra x)) \arr (\text{Eql}(B) x y)}} &
    \multicolumn{2}{r}{
    \inferrule[eql]
    {B : A \arr \sort{u}}
    {\text{eql}(B) : (x : !(A)) \arr (y : (!\la B \ra x)) \arr \sort{0}}} \\\\
    \multicolumn{2}{l}{
    \inferrule[Comp]
    {g : (x : !(A)) \arr ((y : A) \times B y), f : (x : ((x : A) \times B x)) \arr C x}
    {f \circ g : (x : !(A)) \arr (C \circ g) x}} &
    \multicolumn{1}{r}{
    \inferrule[Comp']
    {g : A \arr B, f : (x : B) \arr C x}
    {f \circ ' g : (x : A) \arr (C \circ ' g) x}} \\\\
    \multicolumn{2}{l}{
    \inferrule[Comp2]
    {g : (x : !(A)) \arr ((y : A) \times B x), f : (x : ((y : A) \times B x)) \arr C x}
    {f \circ \circ g : (x : !(A)) \arr (C \circ \circ g) x}} &
    \multicolumn{1}{r}{
    \inferrule[Coup]
    {A : \sort{u}, B : A \arr \sort{v}}
    {\text{Coup}(B) : \sort{max(u, v)}}} \\\\
    \multicolumn{1}{l}{
    \inferrule[Fstc]
    {B : A \arr \sort{v}}
    {\text{Fstc}(B) : \text{Coup}(B) \arr A}
    } &
    \multicolumn{2}{r}{
    \inferrule[Sndc]
    {B : A \arr \sort{v}}
    {\text{Sndc}(B) : (x : \text{Coup}(B)) \arr B (\text{Fstc}(B) x)}}
  \end{tabular}
